checks=[]
def run(name,args,cwd=WT,expected=0):
    env=dict(os.environ,PYTHONPATH=os.pathsep.join([str(cwd/'src'),str(cwd/'tests')]),HF_HUB_OFFLINE='1',TRANSFORMERS_OFFLINE='1',HF_DATASETS_OFFLINE='1',AETHON_OFFLINE_AUDIT_OUTPUT=str(OUT/(name+'-network.json')))
    p=subprocess.run(args,cwd=cwd,env=env,capture_output=True,timeout=900)
    raw=p.stdout+p.stderr
    (OUT/(name+'.txt')).write_bytes(raw)
    checks.append({'name':name,'command':args,'cwd':str(cwd),'exit_code':p.returncode,'expected_exit_code':expected,'status':'PASS' if p.returncode==expected else 'FAIL','sha256':hashlib.sha256(raw).hexdigest()})
    (OUT/'checks.json').write_text(json.dumps(checks,indent=2)+'\n',encoding='utf-8')
    print(name,checks[-1]['status'],raw.decode(errors='replace')[-1200:],flush=True)
    assert p.returncode==expected
