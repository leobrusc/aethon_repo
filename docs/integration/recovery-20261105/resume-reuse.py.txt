import hashlib
import json
import os
from pathlib import Path
import subprocess
ROOT = Path(__file__).resolve().parents[3]
WT = ROOT/'.worktrees/artifact-guard-fix'
OUT = WT/'docs/integration/recovery-20261105'
def git(*args):
    return subprocess.check_output(['git',*args],cwd=WT)
source = (Path(__file__).parent/'run_recovery.py').read_text()
tail = source[source.index('import pandas as pd'):]
tail = tail.replace("assert hashlib.sha256((WT/name).read_bytes()).hexdigest() == digest, name",
    "assert digest in {hashlib.sha256((WT/name).read_bytes()).hexdigest(), hashlib.sha256(git('show','HEAD:'+name)).hexdigest()}, name")
tail = tail.replace("assert git('show','HEAD:'+p.relative_to(WT).as_posix()) == p.read_bytes()",
    "assert json.loads(git('show','HEAD:'+p.relative_to(WT).as_posix())) == item; upstream = p.with_name(p.name.replace('.summary.json','.upstream.json')); assert item['upstream_and_dependency_audit']['raw_sha256'] in {hashlib.sha256(upstream.read_bytes()).hexdigest(), hashlib.sha256(upstream.read_bytes().replace(b'\\r\\n',b'\\n')).hexdigest()}")
if not (OUT/'scientific-reuse.json').exists():
    exec(compile(tail, 'recovery_reuse_continuation', 'exec'))
manifest = json.loads((WT/'docs/validation/VALIDATION_MANIFEST.json').read_text())
checked = []
for field in ['evidence_files','prior_evidence_files']:
    for item in manifest[field]:
        name = item['file']
        location = WT/name
        if not location.exists():
            location = ROOT/'.worktrees/validation'/name
        assert hashlib.sha256(location.read_bytes()).hexdigest() == item['sha256'], name
        checked.append(name)
original = json.loads((WT/'docs/modernization/preserved-local-state.json').read_text())
assert all(hashlib.sha256((ROOT/name).read_bytes()).hexdigest() == digest for name,digest in original.items())
report = {'status':'PASS','verified_validation_evidence_files':len(checked),
    'original_five_files':'PASS','canonical_git_implementation_hashes':'PASS',
    'note':'Initial raw implementation hash assertion failed on generation.py CRLF checkout. Continuation compares recorded Git bytes as well; no source change.',
    'git_locks':[str(p.relative_to(ROOT)) for p in (ROOT/'.git').rglob('*.lock')],
    'project_session_directories':{name:(ROOT/name).exists() for name in ['.codex','.agents']}}
assert not report['git_locks']
(OUT/'continuation.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
print(json.dumps(report,indent=2))
