from pathlib import Path

import json

from datetime import datetime,timezone
ROOT=Path(__file__).resolve().parents[3]
WT=ROOT/'.worktrees/ed25519-proposal'
PY=str(ROOT/'.worktrees/security/.venv-security/Scripts/python.exe')
OUT=WT/'results/audit'/('ed25519-proposal-'+datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ'))
OUT.mkdir(parents=True,exist_ok=False)
ENV=dict(os.environ,PYTHONPATH=os.pathsep.join([str(WT/'src'),str(WT/'tests')]),

checks=[]
def run(name,args,expected=0):
    p=subprocess.run(args,cwd=WT,env=ENV,capture_output=True,timeout=900)
    raw=p.stdout+p.stderr
    (OUT/(name+'.txt')).write_bytes(raw)
    checks.append({'name':name,'command':args,'cwd':str(WT),'exit_code':p.returncode,'expected_exit_code':expected,
                   'status':'PASS' if p.returncode==expected else 'FAIL','sha256':hashlib.sha256(raw).hexdigest()})
    (OUT/'checks.json').write_text(json.dumps(checks,indent=2)+'\n',encoding='utf-8')
    print(name,checks[-1]['status'],raw.decode(errors='replace')[-1500:],flush=True)
    if p.returncode!=expected: raise SystemExit(1)
