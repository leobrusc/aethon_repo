"""Read public package metadata only. Never request wheel/sdist payloads."""
from pathlib import Path
from datetime import datetime,timezone
import urllib.request,json,hashlib,sys,platform,importlib.util
ROOT=Path(__file__).resolve().parents[3]
WT=ROOT/'.worktrees/ed25519-proposal'
OUT=WT/'results/audit'/('ed25519-metadata-'+datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ'))
OUT.mkdir(parents=True,exist_ok=False)
records=[]
def fetch(name,url):
    assert url.startswith('https://pypi.org/pypi/') and url.endswith('/json')
    with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'AETHON-metadata-review/1'}),timeout=30) as r:
        raw=r.read(2*1024*1024)
    (OUT/(name+'.json')).write_bytes(raw)
    records.append({'name':name,'url':url,'sha256':hashlib.sha256(raw).hexdigest(),'scope':'JSON metadata only; no distribution downloaded'})
    return json.loads(raw)
allmeta={}
for name,path in [('cryptography','50.0.2/'),('cffi',''),('pycparser','')]:
    data=fetch(name,'https://pypi.org/pypi/'+name+'/'+path+'json')
    allmeta[name]=data
    info=data['info']
    wheels=[f for f in data['urls'] if f['packagetype']=='bdist_wheel' and any(t in f['filename'] for t in ['win_amd64','manylinux_2_28_x86_64','manylinux2014_x86_64','py3-none-any'])]
    print(json.dumps({'name':name,'version':info['version'],'requires_python':info.get('requires_python'),'requires_dist':info.get('requires_dist'),'license':info.get('license_expression') or info.get('license'),'vulnerabilities':data.get('vulnerabilities'),'files':[{'filename':f['filename'],'sha256':f['digests']['sha256'],'size':f['size'],'upload_time':f['upload_time_iso_8601'],'yanked':f['yanked'],'url':f['url']} for f in wheels]},indent=2),flush=True)
runtime={'python':sys.version,'executable':sys.executable,'base_executable':sys._base_executable,'cache_tag':sys.implementation.cache_tag,'platform':platform.platform(),'architecture':platform.machine(),'packages':{n:importlib.util.find_spec(n) is not None for n in ['cryptography','nacl','cffi','pycparser']}}
for name in ['executable','base_executable']:
    p=Path(runtime[name]); runtime[name+'_sha256']=hashlib.sha256(p.read_bytes()).hexdigest()
(OUT/'runtime-inventory.json').write_text(json.dumps(runtime,indent=2)+'\n',encoding='utf-8')
(OUT/'sources.json').write_text(json.dumps(records,indent=2)+'\n',encoding='utf-8')
print('METADATA OUTPUT '+str(OUT),flush=True)
